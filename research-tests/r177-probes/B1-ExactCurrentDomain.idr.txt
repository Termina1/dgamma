module DGamma.R177ExactCurrentDomainProbe

import DGamma.Calculus
import DGamma.CP3
import DGamma.CP5UniqueRawNameInsertions
import Decidable.Equality

%default total
%unbound_implicits off

||| One exact supported current generation crosses the two ACCEPTED scans in
||| either direction. This does not identify a scanner stamp with a located
||| insertion; uniqueness must not be applied until that separate authentication.
0 r177ExactCurrentSupportedDomain :
  (name, key, world, error : Type) -> (value : key -> Type) ->
  (nameEq : DecEq name) -> (keyEq : DecEq key) ->
  {initial, leftFinal, rightFinal : SystemState name key value world error} ->
  (leftTrace : Transitions initial leftFinal) ->
  (rightTrace : Transitions initial rightFinal) ->
  (sameInputs : SameOrchestrationModuloGenerated nameEq keyEq leftTrace rightTrace) ->
  (leftUnique : UniqueRawNameInsertions name key world error value nameEq keyEq leftTrace) ->
  (rightUnique : UniqueRawNameInsertions name key world error value nameEq keyEq rightTrace) ->
  (((selected : name) -> (generation : RegistrationGeneration name) ->
    (isSupported @{nameEq} @{keyEq} {key = key} {value = value} {world = world} {error = error} selected leftFinal = True) ->
    (lookupCurrentGeneration @{nameEq} selected (leftFinalGenerations (generatedRegistrationTree sameInputs)) = Just generation) ->
    (mapped : RegistrationGeneration name **
      (generationForward (generatedGenerationBijection sameInputs) generation = mapped,
       lookupCurrentGeneration @{nameEq} (renameForward (currentNameBijection (endpointRenaming sameInputs)) selected)
         (rightFinalGenerations (generatedRegistrationTree sameInputs)) = Just mapped))),
   ((selected : name) -> (generation : RegistrationGeneration name) ->
    (isSupported @{nameEq} @{keyEq} {key = key} {value = value} {world = world} {error = error} selected rightFinal = True) ->
    (lookupCurrentGeneration @{nameEq} selected (rightFinalGenerations (generatedRegistrationTree sameInputs)) = Just generation) ->
    (mapped : RegistrationGeneration name **
      (generationBackward (generatedGenerationBijection sameInputs) generation = mapped,
       lookupCurrentGeneration @{nameEq} (renameBackward (currentNameBijection (endpointRenaming sameInputs)) selected)
         (leftFinalGenerations (generatedRegistrationTree sameInputs)) = Just mapped))))
r177ExactCurrentSupportedDomain name key world error value nameEq keyEq leftTrace rightTrace sameInputs leftUnique rightUnique =
  ( (\selected, generation, supported, current =>
      case leftCurrentGenerationMapped (endpointRenaming sameInputs) selected generation current of
        Left vestigial => case trans (sym supported) (vestigialUnsupported vestigial) of Refl impossible
        Right mapped => mapped)
  , (\selected, generation, supported, current =>
      case rightCurrentGenerationMapped (endpointRenaming sameInputs) selected generation current of
        Left vestigial => case trans (sym supported) (vestigialUnsupported vestigial) of Refl impossible
        Right mapped => mapped))
